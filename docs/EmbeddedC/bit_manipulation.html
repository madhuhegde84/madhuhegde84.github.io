Bit Manipulation in Embedded C

A quick reference guide for firmware development — covering essential bit operations, register programming patterns, and hardware-level optimizations.

1. Core Bitwise Operators

Understanding the fundamental bitwise operators is critical for efficient register and hardware manipulation in embedded systems.

Operator Symbol Example Description Firmware Use Case
AND & 0b1101 & 0b1011 = 0b1001 Sets bit to 1 only if both bits are 1 Masking / clearing bits
OR | 0b1100 | 0b0011 = 0b1111 Sets bit to 1 if either bit is 1 Setting bits
XOR ^ 0b1010 ^ 0b0110 = 0b1100 Sets bit to 1 if bits are different Toggling bits, swap without temp
NOT ~ ~0b1010 = 0b0101 (inverted) Inverts all bits Creating inverse masks

2. Essential Bit Manipulation Macros

Standard macros used across embedded firmware to manipulate individual bits safely and efficiently.

Single Bit Operations

/* Define bit position helper */
#define BIT(n)           (1U << (n))

/* Set the n-th bit in register */
#define SET_BIT(reg, n)     ((reg) |= BIT(n))

/* Clear the n-th bit in register */
#define CLEAR_BIT(reg, n)   ((reg) &= ~BIT(n))

/* Toggle the n-th bit in register */
#define TOGGLE_BIT(reg, n)  ((reg) ^= BIT(n))

/* Test if n-th bit is set (returns 0 or 1) */
#define TEST_BIT(reg, n)    (((reg) >> (n)) & 1U)

/* Example usage: Enable UART transmitter (bit 3) */
SET_BIT(UART_CTRL, 3);

/* Check if RX buffer full (bit 5) */
if (TEST_BIT(UART_STATUS, 5)) {
    process_rx_data();
}
Important: Use 1U (unsigned) to avoid sign extension issues on 32-bit systems. Shifting signed integers can cause undefined behavior.

Multi-Bit Field Operations

/* Extract a multi-bit field from register */
#define FIELD_GET(reg, mask, shift) \  
    (((reg) & (mask)) >> (shift))

/* Insert a value into a multi-bit field */
#define FIELD_SET(reg, mask, shift, val) \
    ((reg) = ((reg) & ~(mask)) | (((val) << (shift)) & (mask)))

/* Example: SPI control register with mode field at bits [6:4] */
#define SPI_MODE_MASK   (0x7 << 4)  /* 3 bits wide */
#define SPI_MODE_SHIFT  4

/* Set SPI mode to 3 */
uint32_t ctrl = SPI_CTRL;
FIELD_SET(ctrl, SPI_MODE_MASK, SPI_MODE_SHIFT, 3);
SPI_CTRL = ctrl;

/* Read current SPI mode */
uint8_t mode = FIELD_GET(SPI_CTRL, SPI_MODE_MASK, SPI_MODE_SHIFT);

3. Classic Bit Manipulation Tricks

These patterns appear frequently in embedded firmware and are essential for optimization.

Check if Power of 2

/* Returns 1 if x is power of 2, else 0 */
int is_power_of_2(uint32_t x) {
    return x && !(x & (x - 1));
}

/* Why it works: Power of 2 has exactly one bit set.
   x-1 flips all bits after that bit.
   Example: x=8 (0b1000), x-1=7 (0b0111), x&(x-1)=0 */

/* Firmware use: Validating buffer sizes for DMA alignment */
if (!is_power_of_2(dma_buffer_size)) {
    return ERROR_INVALID_ALIGNMENT;
}

Count Set Bits (Popcount / Hamming Weight)

/* Brian Kernighan's algorithm - O(number of set bits) */
int popcount(uint32_t x) {
    int count = 0;
    while (x) {
        x &= (x - 1);  /* Clears the least significant set bit */
        count++;
    }
    return count;
}

/* Use case: Check how many interrupts are pending */
if (popcount(IRQ_STATUS) > 5) {
    /* Interrupt storm detected */
    disable_peripheral();
}

Isolate Lowest Set Bit

/* Get only the lowest set bit */
uint32_t lsb = x & (-x);

/* Example: x = 0b11010000
            -x = 0b00110000 (two's complement)
       x & -x = 0b00010000 (isolates lowest 1) */

/* Clear lowest set bit */
x = x & (x - 1);

/* Firmware pattern: Process pending interrupts one at a time */
while (irq_pending) {
    uint32_t bit = irq_pending & (-irq_pending);
    handle_interrupt(bit);
    irq_pending &= (irq_pending - 1);  /* Clear handled */
}

XOR Swap (No Temporary Variable)

/* Swap two variables without temp */
void xor_swap(uint32_t *a, uint32_t *b) {
    if (a != b) {  /* Critical: avoid swapping same location */
        *a ^= *b;
        *b ^= *a;
        *a ^= *b;
    }
}

/* Use case: Byte order reversal in packet processing */
Key insight: x & (x-1) clears the rightmost set bit. This is the foundation of many efficient bit manipulation algorithms.

4. Byte Manipulation & Endianness

Essential for packet processing, network protocols, and hardware descriptor programming.

Byte Swap (Endian Conversion)

/* Swap bytes in 16-bit value */
uint16_t swap16(uint16_t x) {
    return (x >> 8) | (x << 8);
}

/* Swap bytes in 32-bit value */
uint32_t swap32(uint32_t x) {
    return ((x >> 24) & 0xFF)       |
           ((x >> 8)  & 0xFF00)     |
           ((x << 8)  & 0xFF0000)   |
           ((x << 24) & 0xFF000000);
}

/* Modern approach: Use compiler builtins */
#define BSWAP16(x) __builtin_bswap16(x)
#define BSWAP32(x) __builtin_bswap32(x)

/* 802.11 MAC header example (network byte order) */
typedef struct __attribute__((packed)) {
    uint16_t frame_ctrl;  /* Little-endian on most SoCs */
    uint16_t duration;
    uint8_t  addr1[6];
} wlan_hdr_t;

Extract and Insert Bytes

/* Extract byte at position n (0 = LSB) */
#define BYTE_GET(x, n)  (((x) >> ((n) * 8)) & 0xFF)

/* Insert byte at position n */
#define BYTE_SET(x, n, val) \
    ((x) = ((x) & ~(0xFF << ((n)*8))) | (((val) & 0xFF) << ((n)*8)))

/* Example: Extract RSSI from PHY status word */
uint32_t phy_stat = 0x12345678;
uint8_t rssi = BYTE_GET(phy_stat, 0);  /* = 0x78 */
WLAN firmware tip: 802.11 headers are in network byte order (big-endian). On little-endian SoCs (ARM Cortex-M, most MIPS), always use ntohs() / htons() or manual swaps when parsing sequence numbers, duration fields, etc.

5. Register Programming Patterns

Safe and efficient patterns for memory-mapped I/O and peripheral control.

Write-1-to-Clear (W1C) Interrupt Status

/* Many SoC interrupt status registers use W1C:
   Writing 1 clears the bit; writing 0 has no effect.
   Reading returns current pending status. */

#define IRQ_RX_DONE   BIT(0)
#define IRQ_TX_DONE   BIT(1)
#define IRQ_ERROR     BIT(7)

void handle_interrupts(void) {
    uint32_t status = IRQ_STATUS_REG;
    
    /* WRONG: Read-modify-write can lose interrupts */
    /* IRQ_STATUS_REG &= ~IRQ_RX_DONE;  // DON'T DO THIS */
    
    /* CORRECT: Write only the bits you want to clear */
    IRQ_STATUS_REG = IRQ_RX_DONE | IRQ_TX_DONE;
    
    /* Process each interrupt */
    if (status & IRQ_RX_DONE)  handle_rx();
    if (status & IRQ_TX_DONE)  handle_tx();
    if (status & IRQ_ERROR)    handle_error();
}
Critical pattern: For W1C registers, write back exactly the bits you want to clear. Never use &= ~mask as it may unintentionally clear newly-arrived interrupts.

Atomic Read-Modify-Write

/* Problem: RMW is NOT atomic - interrupts can corrupt state */

/* UNSAFE in ISR context */
void unsafe_set_bit(volatile uint32_t *reg, uint8_t n) {
    *reg |= BIT(n);  /* Read, modify, write - 3 bus cycles */
}

/* SAFE: Disable interrupts for critical section */
void safe_set_bit(volatile uint32_t *reg, uint8_t n) {
    uint32_t irq_state = disable_global_irq();
    *reg |= BIT(n);
    restore_global_irq(irq_state);
}

/* For multi-core: Use hardware atomic operations */
__sync_fetch_and_or(reg, BIT(n));  /* GCC builtin */

Bitmask Flags with Iteration

/* Efficient pattern for processing multiple set bits */
#define FLAG_A  BIT(0)
#define FLAG_B  BIT(3)
#define FLAG_C  BIT(7)

void process_flags(uint32_t flags) {
    /* Process each set bit */
    while (flags) {
        uint32_t bit = flags & (-flags);  /* Isolate LSB */
        int pos = __builtin_ctz(bit);     /* Count trailing zeros */
        
        switch (pos) {
            case 0: handle_flag_a(); break;
            case 3: handle_flag_b(); break;
            case 7: handle_flag_c(); break;
        }
        
        flags &= (flags - 1);  /* Clear processed bit */
    }
}

6. Memory-Mapped I/O Access Patterns

Defining Hardware Registers

/* Method 1: Direct pointer cast */
#define UART_BASE       0x40011000U
#define UART_SR         (*(volatile uint32_t *)(UART_BASE + 0x00))
#define UART_DR         (*(volatile uint32_t *)(UART_BASE + 0x04))
#define UART_CTRL       (*(volatile uint32_t *)(UART_BASE + 0x08))

/* Method 2: Register structure (preferred for peripheral blocks) */
typedef struct {
    volatile uint32_t SR;      /* Status register */
    volatile uint32_t DR;      /* Data register */
    volatile uint32_t CTRL;    /* Control register */
    volatile uint32_t BRR;     /* Baud rate register */
} UART_TypeDef;

#define UART  ((UART_TypeDef *)UART_BASE)

/* Usage */
UART->CTRL |= BIT(0);  /* Enable UART */
while (!(UART->SR & BIT(7)))  /* Wait for TX empty */
    ;
UART->DR = 'A';  /* Send character */
volatile required: Always use volatile for hardware registers. The compiler must generate actual memory reads/writes and not cache values in CPU registers.

7. Advanced Bit Manipulation Techniques

Less commonly asked but critical for firmware optimization and problem-solving.

Reverse Bits

/* Reverse bits in a 32-bit integer */
uint32_t reverse_bits_loop(uint32_t x) {
    uint32_t result = 0;
    for (int i = 0; i < 32; i++) {
        result = (result << 1) | (x & 1);
        x >>= 1;
    }
    return result;
}

/* Divide-and-conquer approach (faster) */
uint32_t reverse_bits(uint32_t x) {
    x = ((x >> 1)  & 0x55555555) | ((x & 0x55555555) << 1);   /* Swap pairs */
    x = ((x >> 2)  & 0x33333333) | ((x & 0x33333333) << 2);   /* Swap nibbles */
    x = ((x >> 4)  & 0x0F0F0F0F) | ((x & 0x0F0F0F0F) << 4);   /* Swap bytes */
    x = ((x >> 8)  & 0x00FF00FF) | ((x & 0x00FF00FF) << 8);   /* Swap 16-bit */
    x = ((x >> 16) & 0x0000FFFF) | ((x & 0x0000FFFF) << 16);  /* Swap halves */
    return x;
}

/* Use case: Bit-reversed addressing in FFT, CRC table generation */

Parallel Popcount (Constant Time)

/* Count set bits in O(1) - no loops */
int popcount_parallel(uint32_t x) {
    x = x - ((x >> 1) & 0x55555555);
    x = (x & 0x33333333) + ((x >> 2) & 0x33333333);
    x = (x + (x >> 4)) & 0x0F0F0F0F;
    x = x + (x >> 8);
    x = x + (x >> 16);
    return x & 0x3F;
}

/* Modern alternative: Use GCC builtin */
#define POPCOUNT(x) __builtin_popcount(x)

8. Medium to Advanced Bit Manipulation Problems Advanced

These problems test deeper algorithmic thinking in embedded and firmware contexts.

Problem 1: Find Two Non-Repeating Numbers Using XOR

Given an array where all elements repeat twice except two — find those two elements using O(1) space.

/* Concept: XOR all elements. Result = a XOR b (the two non-repeating nums).
   Find a set bit in result to partition array into two groups.
   XOR each group independently to get a and b. */
void find_two_unique(int *arr, int n, int *a, int *b) {
    int xor_all = 0;
    for (int i = 0; i < n; i++) xor_all ^= arr[i];
    
    /* Find any set bit (rightmost) in xor_all */
    int set_bit = xor_all & (-xor_all);
    
    *a = 0; *b = 0;
    for (int i = 0; i < n; i++) {
        if (arr[i] & set_bit)
            *a ^= arr[i];  /* Group 1: bit is set */
        else
            *b ^= arr[i];  /* Group 2: bit is clear */
    }
}

/* Firmware use: Finding two faulty DMA channels or IRQ IDs */

Problem 2: Gray Code Encode and Decode

Gray code changes by only one bit between successive values. Used in rotary encoders and ADC communication.

/* Binary to Gray code */
uint32_t bin_to_gray(uint32_t n) {
    return n ^ (n >> 1);
}

/* Gray to Binary */
uint32_t gray_to_bin(uint32_t g) {
    uint32_t n = 0;
    for (; g; g >>= 1)
        n ^= g;
    return n;
}

/* Example:
   Binary  0001 = 1  ->  Gray 0001
   Binary  0010 = 2  ->  Gray 0011
   Binary  0011 = 3  ->  Gray 0010
   Only one bit changes each step */

/* Firmware use: RF channel selector with mechanical encoder */

Problem 3: Find Missing Number in [0..N]

Given an array of N numbers from range [0..N] with one missing, find it.

/* Approach 1: XOR (most robust, no overflow) */
uint32_t find_missing_xor(uint32_t *arr, int n) {
    uint32_t xor_all = n;  /* XOR with N itself */
    for (int i = 0; i < n; i++) {
        xor_all ^= i ^ arr[i];
    }
    return xor_all;
}

/* Approach 2: Sum formula (simpler but can overflow for large N) */
uint32_t find_missing_sum(uint32_t *arr, int n) {
    uint64_t expected = (uint64_t)n * (n + 1) / 2;
    uint64_t actual = 0;
    for (int i = 0; i < n; i++) actual += arr[i];
    return (uint32_t)(expected - actual);
}

/* Firmware use: Detect missing sequence numbers in packet stream */

Problem 4: Next Power of Two (OR-Fold Technique)

Round up a value to the next power of 2 — critical for buffer alignment and DMA descriptor sizing.

uint32_t next_pow2(uint32_t x) {
    if (x == 0) return 1;
    x--;           /* Handle exact powers of 2 */
    x |= x >> 1;  /* Smear highest set bit downward */
    x |= x >> 2;
    x |= x >> 4;
    x |= x >> 8;
    x |= x >> 16;
    return x + 1;
}

/* How it works:
   x=6 (0b00000110)
   x-- = 5 (0b00000101)
   After OR-folds: 0b00000111 (all 1s below MSB)
   +1 = 8 (0b00001000) */

/* Use case: Allocate DMA buffer that must be 2^n aligned */
size_t aligned_size = next_pow2(requested_size);

Problem 5: Swap Bits at Positions i and j

void swap_bits(uint32_t *x, int i, int j) {
    /* XOR trick: only swap if bits are different */
    uint32_t bi = (*x >> i) & 1;
    uint32_t bj = (*x >> j) & 1;
    
    if (bi != bj) {  /* Only toggle if different */
        *x ^= (1U << i) | (1U << j);
    }
}

/* Why XOR works:
   If bi==bj, XOR toggles both to opposite values - wrong!
   Only toggle when they differ - hence the check. */

/* Use case: Reorder bits in hardware descriptor flags */

Problem 6: Bitmap-Based Resource Allocator

A lock-free bitmask allocator for DMA channels, memory pools, or hardware resources.

#define MAX_CHANNELS    32
static volatile uint32_t channel_bitmap = 0xFFFFFFFF;  /* 1 = free */

/* Allocate a free channel, returns -1 if none available */
int alloc_channel(void) {
    uint32_t bmap, bit;
    do {
        bmap = channel_bitmap;
        if (bmap == 0) return -1;  /* All busy */
        bit = bmap & (-bmap);      /* Isolate lowest free channel */
    } while (!__sync_bool_compare_and_swap(
                &channel_bitmap, bmap, bmap & ~bit));
    return __builtin_ctz(bit);     /* Return channel index */
}

/* Free a channel */
void free_channel(int ch) {
    __sync_fetch_and_or(&channel_bitmap, 1U << ch);
}

/* Check channel status */
int is_channel_busy(int ch) {
    return !(channel_bitmap & (1U << ch));
}

Problem 7: Circular Ring Buffer with AND-Wrap

Using bitwise AND for ring buffer wrap-around (valid only for power-of-2 sizes). This avoids modulo which can be expensive on embedded MCUs without hardware divider.

#define RING_SIZE   64     /* Must be power of 2 */
#define RING_MASK   (RING_SIZE - 1)

typedef struct {
    volatile uint32_t  prod_idx;  /* Producer (firmware) */
    volatile uint32_t  cons_idx;  /* Consumer (hardware) */
    tx_desc_t          ring[RING_SIZE];
} ring_t;

/* Get number of used slots */
static inline uint32_t ring_count(ring_t *r) {
    return (r->prod_idx - r->cons_idx) & RING_MASK;
}

/* Get number of free slots */
static inline uint32_t ring_free(ring_t *r) {
    return RING_SIZE - ring_count(r) - 1;
}

/* Enqueue descriptor */
int ring_enqueue(ring_t *r, tx_desc_t *desc) {
    if (ring_free(r) == 0) return -1;  /* Full */
    
    r->ring[r->prod_idx & RING_MASK] = *desc;
    
    /* Memory barrier before updating producer index */
    __DMB();  /* Data Memory Barrier (ARM) */
    r->prod_idx++;
    return 0;
}

/* Dequeue descriptor */
int ring_dequeue(ring_t *r, tx_desc_t *desc) {
    if (ring_count(r) == 0) return -1;  /* Empty */
    
    *desc = r->ring[r->cons_idx & RING_MASK];
    __DMB();
    r->cons_idx++;
    return 0;
}

/* Why AND instead of modulo:
   (prod % RING_SIZE) requires division (slow on MCUs)
   (prod & RING_MASK) is a single AND instruction */

Problem 8: Multiply / Divide Without Hardware Multiplier

Some microcontrollers lack hardware multiply/divide. Shift-and-add implements it efficiently.

/* Multiply x by n using shifts and adds */
uint32_t mul_by_shifts(uint32_t x, uint32_t n) {
    uint32_t result = 0;
    while (n) {
        if (n & 1) result += x;  /* Add current power */
        x <<= 1;                 /* Double x */
        n >>= 1;                 /* Halve n */
    }
    return result;
}

/* Divide x by constant n using multiplication by magic constant */
/* For divide by 10: multiply by magic number and shift right */
uint32_t div_by_10(uint32_t x) {
    /* magic = ceil(2^35 / 10) = 0xCCCCCCCCUL >> adjustment */
    return (uint32_t)(((uint64_t)x * 0xCCCCCCCDULL) >> 35);
}

/* Multiply by non-power-of-2 constants */
uint32_t mul_by_10(uint32_t x) {
    return (x << 3) + (x << 1);  /* x*8 + x*2 = x*10 */
}

uint32_t mul_by_3(uint32_t x) {
    return x + (x << 1);         /* x + x*2 = x*3 */
}

Problem 9: Rotate Bits Left/Right (CRC step)

/* Rotate left: bits shifted out from MSB reappear at LSB */
uint32_t rotl32(uint32_t x, uint8_t n) {
    n &= 31;  /* Safety: modulo 32 */
    return (x << n) | (x >> (32 - n));
}

/* Rotate right */
uint32_t rotr32(uint32_t x, uint8_t n) {
    n &= 31;
    return (x >> n) | (x << (32 - n));
}

/* CRC-32 one-step using rotate (bit-by-bit) */
#define CRC32_POLY  0xEDB88320U  /* Reflected polynomial */

uint32_t crc32_step(uint32_t crc, uint8_t byte) {
    crc ^= byte;
    for (int i = 0; i < 8; i++) {
        uint32_t lsb = crc & 1;
        crc >>= 1;
        if (lsb) crc ^= CRC32_POLY;  /* XOR with poly if bit was set */
    }
    return crc;
}

/* Firmware use: 802.11 FCS (Frame Check Sequence) validation */
